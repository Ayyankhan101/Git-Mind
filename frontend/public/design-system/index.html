export function validateToken(token: string) {
  const decoded = jwt.verify(token, SECRET);
  const now = Math.floor(Date.now() / 1000);

  // BUG: uses strict less-than, should be <=
  if (decoded.exp < now) {
    throw new AuthError('Token expired');
  }

  return decoded;
